2. Codice ponte per il tuo Google Sheet (Estensioni → Apps Script) 
function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheetName = e.parameter.sheet || 'Schede';
  const sheet = ss.getSheetByName(sheetName);
  if (!sheet) return jsonOut({ error: 'Foglio non trovato: ' + sheetName });
  
  const data = sheet.getDataRange().getValues();
  const headers = data.shift();
  const rows = data.map(r => {
    let obj = {};
    headers.forEach((h, i) => obj[h] = r[i]);
    return obj;
  });
  return jsonOut({ status: 'ok', data: rows });
}

function doPost(e) {
  // LockService evita conflitti se più utenti salvano contemporaneamente
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const payload = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(payload.sheet);
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    const newRow = headers.map(h => payload.row[h] !== undefined ? payload.row[h] : '');
    sheet.appendRow(newRow);
    return jsonOut({ status: 'saved' });
  } finally {
    lock.releaseLock();
  }
}

function jsonOut(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

3. Pubblicazione Gratuita su GitHub Pages (Zero Crediti e Zero Badge)

Salva questo file come index.html, crea un repository pubblico gratuito su github.com, trascina il file con Add file → Upload files e attiva Settings → Pages → Branch: main. Avrai un link HTTPS pulito al 100%, senza alcun logo o barra esterna, installabile su smartphone come App nativa.
